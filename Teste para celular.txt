import re
from kivy.app import App
from kivy.uix.boxlayout import BoxLayout
from kivy.uix.scrollview import ScrollView
from kivy.uix.label import Label
from kivy.uix.textinput import TextInput
from kivy.uix.spinner import Spinner
from kivy.uix.button import Button
from kivy.core.window import Window

# Tabela AWG exata (Bitola: Área em mm²)
TABELA_AWG = {
    10: 5.26, 11: 4.17, 12: 3.31, 13: 2.62, 14: 2.08,
    15: 1.65, 16: 1.31, 17: 1.04, 18: 0.823, 19: 0.653,
    20: 0.519, 21: 0.411, 22: 0.326, 23: 0.259, 24: 0.205,
    25: 0.162, 26: 0.129, 27: 0.102, 28: 0.080, 29: 0.064,
    30: 0.051
}

class SubstitutorApp(App):
    def build(self):
        self.title = "Substitutor & Dimensionador Pro"
        
        # Layout principal com rolagem para caber em qualquer celular
        root = ScrollView()
        layout = BoxLayout(orientation='vertical', padding=20, spacing=15, size_hint_y=None)
        layout.bind(minimum_height=layout.setter('height'))
        
        # 1. Fios Originais
        layout.add_widget(Label(text="[b]1. Fios Originais do Motor[/b]\nExemplo: 15x20 ou 4x18 + 2x20", markup=True, size_hint_y=None, height=40))
        self.entry_orig = TextInput(hint_text="Digite aqui...", multiline=False, size_hint_y=None, height=45)
        layout.add_widget(self.entry_orig)
        
        # 2. Configurações de Material e Ligação
        layout.add_widget(Label(text="[b]2. De / Para (Materiais):[/b]", markup=True, size_hint_y=None, height=30))
        box_mat = BoxLayout(orientation='horizontal', spacing=10, size_hint_y=None, height=45)
        self.spin_de = Spinner(text="Alumínio", values=("Cobre", "Alumínio"))
        self.spin_para = Spinner(text="Cobre", values=("Cobre", "Alumínio"))
        box_mat.add_widget(self.spin_de)
        box_mat.add_widget(self.spin_para)
        layout.add_widget(box_mat)
        
        layout.add_widget(Label(text="[b]Tipo de Ligação Elétrica:[/b]", markup=True, size_hint_y=None, height=30))
        self.spin_ligacao = Spinner(
            text="Série (Circuito Único)", 
            values=("Série (Circuito Único)", "Paralelo (2 Circuitos)", "Paralelo (4 Circuitos)"),
            size_hint_y=None, height=45
        )
        layout.add_widget(self.spin_ligacao)
        
        # 3. Fios Novos
        layout.add_widget(Label(text="[b]3. Fios Novos de Substituição[/b]\nExemplo: 15x22 ou 10x21", markup=True, size_hint_y=None, height=40))
        self.entry_sub = TextInput(hint_text="Digite aqui...", multiline=False, size_hint_y=None, height=45)
        layout.add_widget(self.entry_sub)
        
        # Botão Calcular
        btn = Button(text="CALCULAR REBOBINAGEM", background_color=(0.15, 0.7, 0.4, 1), font_size=16, bold=True, size_hint_y=None, height=55)
        btn.bind(on_press=self.calcular)
        layout.add_widget(btn)
        
        # Painel de Resultados
        layout.add_widget(Label(text="[b][color=3399ff]RESULTADOS TÉCNICOS[/color][/b]", markup=True, size_hint_y=None, height=30))
        self.lbl_resultado = Label(
            text="Preencha os campos acima para simular.", 
            halign="left", valign="top", markup=True, size_hint_y=None, height=180
        )
        self.lbl_resultado.bind(size=self.lbl_resultado.setter('text_size'))
        layout.add_widget(self.lbl_resultado)
        
        root.add_widget(layout)
        return root

    def interpretar_fios(self, texto):
        texto = texto.lower().replace(" ", "").replace(",", ".")
        partes = texto.split("+")
        area_total = 0.0
        total_fios = 0
        
        for parte in partes:
            if not parte: continue
            if "x" in parte: padrao = parte.split("x")
            elif "*" in parte: padrao = parte.split("*")
            else: padrao = ["1", parte]
                
            try:
                qtd = int(padrao[0])
                awg = int(padrao[1])
                if awg in TABELA_AWG:
                    area_total += (TABELA_AWG[awg] * qtd)
                    total_fios += qtd
                else: return 0, 0, True
            except: return 0, 0, True
                
        return area_total, total_fios, False

    def calcular(self, instance):
        area_orig, qtd_orig, erro_orig = self.interpretar_fios(self.entry_orig.text)
        area_sub, qtd_sub, erro_sub = self.interpretar_fios(self.entry_sub.text)
        
        if erro_orig or area_orig == 0 or erro_sub or area_sub == 0:
            self.lbl_resultado.text = "[color=ff3333][b]Erro:[/b] Verifique o formato dos fios digitados.[/color]"
            return
            
        fator = 1.0
        if self.spin_de.text == "Cobre" and self.spin_para.text == "Alumínio": fator = 1.6
        elif self.spin_de.text == "Alumínio" and self.spin_para.text == "Cobre": fator = 1.0 / 1.6
        area_necessaria_total = area_orig * fator
        
        tipo_ligacao = self.spin_ligacao.text
        if tipo_ligacao == "Paralelo (2 Circuitos)": area_necessaria_total /= 2
        elif tipo_ligacao == "Paralelo (4 Circuitos)": area_necessaria_total /= 4
            
        diferenca = ((area_sub - area_necessaria_total) / area_necessaria_total) * 100
        
        densidade = 5.0 if self.spin_para.text == "Cobre" else 3.1
        area_real_motor = area_sub
        if tipo_ligacao == "Paralelo (2 Circuitos)": area_real_motor *= 2
        elif tipo_ligacao == "Paralelo (4 Circuitos)": area_real_motor *= 4
        
        corrente_estimada = area_real_motor * densidade
        
        if corrente_estimada <= 15.5: cabo = "1.5 mm²"
        elif corrente_estimada <= 21: cabo = "2.5 mm²"
        elif corrente_estimada <= 28: cabo = "4.0 mm²"
        elif corrente_estimada <= 36: cabo = "6.0 mm²"
        elif corrente_estimada <= 50: cabo = "10.0 mm²"
        else: cabo = "16.0 mm² ou maior"
            
        cor_erro = "33cc33" if -5.0 <= diferenca <= 5.0 else ("ff9900" if diferenca > 5.0 else "ff3333")
        
        self.lbl_resultado.text = (
            f"• Seção Original: {area_orig:.3f} mm² ({qtd_orig} fios)\n"
            f"• Seção p/ Grupo Requerida: {area_necessaria_total:.3f} mm²\n"
            f"• Seção Nova Proposta: {area_sub:.3f} mm² ({qtd_sub} fios)\n"
            f"• Erro de Escala: [color={cor_erro}][b]{diferenca:+.2f}%[/b][/color]\n"
            f"• Corrente Máx: {corrente_estimada:.1f} A\n"
            f"• Cabo Externo Comercial: [color=3399ff][b]{cabo}[/b][/color]"
        )

if __name__ == '__main__':
    SubstitutorApp().run()
